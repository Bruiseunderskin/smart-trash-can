cmake_minimum_required(VERSION 3.20)
project(smart_trash_can_preview C CXX ASM)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../xiaozhi-esp32/managed_components/lvgl__lvgl" CACHE PATH "Existing LVGL 9.5 source checkout")
if(NOT EXISTS "${LVGL_DIR}/lv_version.h")
    message(FATAL_ERROR "Set LVGL_DIR to an LVGL 9.5 checkout (see README).")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
find_package(SDL2 REQUIRED)
add_subdirectory("${LVGL_DIR}" lvgl)
target_include_directories(lvgl SYSTEM PUBLIC ${SDL2_INCLUDE_DIRS})
file(GLOB UI_GENERATED CONFIGURE_DEPENDS "../generated/*.c")
add_library(stc_ui STATIC ../src/stc_ui.c ${UI_GENERATED})
target_include_directories(stc_ui PUBLIC ../include PRIVATE ../src ../generated)
target_link_libraries(stc_ui PUBLIC lvgl)
target_compile_options(stc_ui PRIVATE -Wall -Wextra -Werror)
add_executable(main src/main.c)
target_link_libraries(main PRIVATE stc_ui ${SDL2_LIBRARIES})
target_compile_options(main PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME state_and_layout COMMAND main --self-test)
set_tests_properties(state_and_layout PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
